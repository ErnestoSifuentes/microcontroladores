#include <msp430.h>

volatile unsigned char rx_data = 0;

void clock_init(void)
{
    BCSCTL1 = CALBC1_1MHZ;
    DCOCTL  = CALDCO_1MHZ;
}

void UART_init(void)
{
    /* P1.1 = RXD, P1.2 = TXD */
    P1SEL  |= BIT1 | BIT2;
    P1SEL2 |= BIT1 | BIT2;

    UCA0CTL1 |= UCSWRST;
    UCA0CTL1 |= UCSSEL_2;      /* SMCLK = 1 MHz */

    /* 9600 baudios para 1 MHz */
    UCA0BR0 = 104;
    UCA0BR1 = 0;
    UCA0MCTL = UCBRS0;

    UCA0CTL1 &= ~UCSWRST;
    IE2 |= UCA0RXIE;
}

void UART_sendChar(unsigned char c)
{
    while (!(IFG2 & UCA0TXIFG));
    UCA0TXBUF = c;
}

void UART_sendString(const char *str)
{
    while (*str)
    {
        UART_sendChar(*str++);
    }
}

int main(void)
{
    WDTCTL = WDTPW | WDTHOLD;

    clock_init();
    UART_init();

    __enable_interrupt();

    UART_sendString("Hola\r\n");

    while (1)
    {
        __bis_SR_register(LPM0_bits + GIE);
        __no_operation();

        if (rx_data == '2')
        {
            UART_sendString("Hola\r\n");
        }
    }
}

#pragma vector=USCIAB0RX_VECTOR
__interrupt void USCI0RX_ISR(void)
{
    if (IFG2 & UCA0RXIFG)
    {
        rx_data = UCA0RXBUF;
        __bic_SR_register_on_exit(LPM0_bits);
    }
}
