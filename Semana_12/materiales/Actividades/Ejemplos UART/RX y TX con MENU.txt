#include <msp430.h>

volatile char rx_data = 0;

void clock_init(void)
{
    BCSCTL1 = CALBC1_1MHZ;
    DCOCTL  = CALDCO_1MHZ;
}

void UART_init(void)
{
    P1SEL  |= BIT1 | BIT2;   // P1.1 = RXD, P1.2 = TXD
    P1SEL2 |= BIT1 | BIT2;

    UCA0CTL1 |= UCSWRST;
    UCA0CTL1 |= UCSSEL_2;    // SMCLK = 1 MHz

    UCA0BR0 = 104;           // 9600 baud
    UCA0BR1 = 0;
    UCA0MCTL = UCBRS0;

    UCA0CTL1 &= ~UCSWRST;

    IE2 |= UCA0RXIE;         // interrupción de recepción
}

void UART_sendChar(char c)
{
    while (!(IFG2 & UCA0TXIFG));
    UCA0TXBUF = c;
}

void UART_sendString(char *str)
{
    while (*str)
    {
        UART_sendChar(*str++);
    }
}

void UART_showMenu(void)
{
    UART_sendString("\r\n===== MENU UART MSP430 =====\r\n");
    UART_sendString("1 -> MSP430 listo\r\n");
    UART_sendString("2 -> Hola\r\n");
    UART_sendString("3 -> Adios\r\n");
    UART_sendString("Seleccione una opcion: \r\n");
}

int main(void)
{
    WDTCTL = WDTPW | WDTHOLD;

    clock_init();
    UART_init();

    P1DIR |= BIT0;           // LED rojo
    P1OUT &= ~BIT0;

    __enable_interrupt();

    UART_showMenu();

    while (1)
    {
        if (rx_data != 0)
        {
            switch (rx_data)
            {
                case '1':
                    UART_sendString("MSP430 listo\r\n");
                    P1OUT ^= BIT0;
                    break;

                case '2':
                    UART_sendString("Hola\r\n");
                    P1OUT ^= BIT0;
                    break;

                case '3':
                    UART_sendString("Adios\r\n");
                    P1OUT ^= BIT0;
                    break;

                default:
                    UART_sendString("Opcion no valida\r\n");
                    break;
            }

            rx_data = 0;
            UART_showMenu();
        }
    }
}

#pragma vector=USCIAB0RX_VECTOR
__interrupt void USCI0RX_ISR(void)
{
    if (IFG2 & UCA0RXIFG)
    {
        rx_data = UCA0RXBUF;
    }
}
